```
# T724: EGL-dispatch probe trigger. Second-stage init parses every /system/etc/init/*.rc
# (proven: installd.rc:109, bootstat.rc:67, dmesgd.rc:4, hw/init.rc:1224 all fired in the
# T723 probe-flight UART). The probe binary lives at /first_stage_ramdisk/t723-probe after
# switch_root (init binary strings prove the path); /t723-probe does not exist post-switch.
# The two `on boot` exec lines are diagnostics: the /first_stage_ramdisk test is EXPECTED to
# succeed, the /t723-probe test is EXPECTED to fail (it probes the pre-switch path). They prove
# .rc parsing + the probe path at guest ~30 s, long before the boot_completed trigger at ~516 s.
service t723probe /first_stage_ramdisk/t723-probe
    class main
    user root
    group root
    oneshot
    disabled

on property:sys.boot_completed=1
    start t723probe

on boot
    exec - root root -- /system/bin/toybox test -x /first_stage_ramdisk/t723-probe
    exec - root root -- /system/bin/toybox test -x /t723-probe
```
